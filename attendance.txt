#include<stdio.h>
int inputAtt(int a[][3]){
    for(int i=0;i<3;i++){
        for(int j=0;j<3;j++){
            scanf("%d",&a[i][j]);
        }
    }
}
int displayAtt(int a[][3]){
    for(int i=0;i<3;i++){
        for(int j=0;j<3;j++){
            printf("%d ",a[i][j]);
        }
    }printf("\n");
}
int totalPre(int a[][3]){
    int count=0;
    for(int i=0;i<3;i++){
        for(int j=0;j<3;j++){
            if(a[i][j]==1){
                count++;
            }
        }
    }printf("Total present are:%d\n",count);
}
int totalAbs(int a[][3]){
    int count=0;
    for(int i=0;i<3;i++){
        for(int j=0;j<3;j++){
            if(a[i][j]==0){
                count++;
            }
        }
    }printf("Total absent are:%d\n",count);
}
int highestAtt(int a[][3]){
    int high=0;
   
    int student=0;
    for(int i=0;i<3;i++){
         int count=0;
        for(int j=0;j<3;j++){
            if(a[i][j]==1){
                count++;
              
            }
        if(count>high){
            high=count;
             student=i;
          }
        
        }
    }printf("highest attents:%d\n",high);
    printf("student is:%d",student);
    printf("\n");
}
int lowerAtt(int a[][3]){
    int lower=4;
    int student=0;
    for(int i=0;i<3;i++){
         int count=0;
        for(int j=0;j<3;j++){
            if(a[i][j]==1){
                count++;
              
            }
        if(count<lower){
            lower=count;
             student=i;
          }
        
        }
    }printf("lower attents:%d\n",lower);
    printf("student is:%d",student);
}
int main(){
    int a[3][3];
    inputAtt(a);
    displayAtt(a);
    totalPre(a);
    totalAbs(a);
    highestAtt(a);
    lowerAtt(a);
}
